90 void Ciudad::comerciar_inventario(Ciudad &c,
91 const map<int, pair<int, int>> &catalogo)
92 {
93  auto it = inventario.begin();
94  auto it2 = c.inventario.begin();
95  while (it != inventario.end() and it2 != c.inventario.end()) {
96  // Si el identificador del producto de la primera ciudad es mas
97  // pequeño, signfica que la segunda ciudad no tiene ese producto, asi
98  // que tenemos que pasar al siguiente producto de la primera ciudad
99  // para comprobar si coincide con el de la segunda
100  if (it->first < it2->first) ++it;
101  // Si el identificador del producto de la segunda ciudad es mas
102  // pequeño, signfica que la primera ciudad no tiene ese producto, asi
103  // que tenemos que pasar al siguiente producto de la segunda ciudad
104  // para comprobar si coincide con el de la primera
105  else if (it->first > it2->first) ++it2;
106  // En el caso que los dos identificadores sean iguales, hemos
107  // encontrado un producto en comun de las dos ciudades
108  else {
109  auto it3 = catalogo.find(it->first);
110  int peso1 = it3->second.first;
111  int vol1 = it3->second.second;
112 
113  auto it4 = catalogo.find(it2->first);
114  int peso2 = it4->second.first;
115  int vol2 = it4->second.second;
116 
117  // La primera ciudad quiere comprar (tiene1 < quiere1) y la segunda
118  // quiere vender (tiene2 > quiere2)
119  if (it->second.first < it->second.second and
120  it2->second.first > it2->second.second) {
121  // La cantidad intercambiada sera la minima que
122  // quieran comprar/vender las ciudades
123  int intercambio = min(it->second.second - it->second.first,
124  it2->second.first - it2->second.second);
125  // Cambios que sufre la primera ciudad
126  it->second.first += intercambio;
127  peso_total += (peso1 * intercambio);
128  volumen_total += (vol1 * intercambio);
129  // Cambios que sufre la segunda ciudad
130  it2->second.first -= intercambio;
131  c.peso_total -= (peso2 * intercambio);
132  c.volumen_total -= (vol2 * intercambio);
133  }
134  // La segunda ciudad quiere comprar (tiene2 < quiere2) y la primera
135  // quiere vender (tiene1 > quiere1)
136  else if (it2->second.first < it2->second.second and
137  it->second.first > it->second.second) {
138  // La cantidad intercambiada sera la minima que
139  // quieran comprar/vender las ciudades
140  int intercambio = min(it2->second.second - it2->second.first,
141  it->second.first - it->second.second);
142  // Cambios que sufre la segunda ciudad
143  it2->second.first += intercambio;
144  c.peso_total += (peso2 * intercambio);
145  c.volumen_total += (vol2 * intercambio);
146  // Cambios que sufre la primera ciudad
147  it->second.first -= intercambio;
148  peso_total -= (peso1 * intercambio);
149  volumen_total -= (vol1 * intercambio);
150  }
151  // Como ya hemos visto el producto que tienen en comun las dos
152  // ciudades, pasamos al siguiente
153  ++it;
154  ++it2;
155  }
156  }
