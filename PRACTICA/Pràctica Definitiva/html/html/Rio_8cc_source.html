84 void Rio::hacer_viaje()
85 {
86  int comprados = embarcacion.consultar_nidcompra();
87  int vendidos = embarcacion.consultar_nidventa();
88  // Ruta contiene el mejor camino posible para el barco
89  list<string> ruta = camino(estructura, infoCiudades, comprados, vendidos,
90  embarcacion.consultar_idcompra(), embarcacion.consultar_idventa());
91 
92  int idc = embarcacion.consultar_idcompra();
93  int idv = embarcacion.consultar_idventa();
94  int nidc = embarcacion.consultar_nidcompra();
95  int nidv = embarcacion.consultar_nidventa();
96  int productosVendidosComprados = 0;
97 
98  for (auto it = ruta.begin(); it != ruta.end(); ++it) {
99  int tiene1 = infoCiudades[*it].consultar_tiene(idc);
100  int quiere1 = infoCiudades[*it].consultar_quiere(idc);
101  int tiene2 = infoCiudades[*it].consultar_tiene(idv);
102  int quiere2 = infoCiudades[*it].consultar_quiere(idv);
103 
104  // Venta de ciudad y compra de barco
105  if (tiene1 > quiere1 and nidc > 0) {
106  // venta contiene la cantidad que puede vender la ciudad sin
107  // dejar al barco en numeros negativos ni a ella misma
108  int venta = min(tiene1 - quiere1, nidc);
109  pair<int, int> aux(tiene1 - venta, quiere1);
110  modificar_producto2(*it, idc, aux);
111  productosVendidosComprados += venta;
112  // Modificamos los productos que ofrece el barco (solo en esta ruta)
113  nidc -= venta;
114  }
115  // Compra de ciudad y venta de barco
116  if (tiene2 < quiere2 and nidv > 0) {
117  // compra contiene la cantidad que puede comprar la ciudad sin
118  // dejar al barco en numeros negativos ni a ella misma
119  int compra = min(quiere2 - tiene2, nidv);
120  pair<int, int> aux(tiene2 + compra, quiere2);
121  modificar_producto2(*it, idv, aux);
122  productosVendidosComprados += compra;
123  // Modificamos los productos que ofrece el barco (solo en esta ruta)
124  nidv -= compra;
125  }
126  }
127 
128  cout << productosVendidosComprados << endl;
129  // Si hay alguna ciudad que comercia anadimos el viaje
130  if (ruta.size() > 0) {
131  auto it = ruta.end();
132  --it;
133  embarcacion.anadir_viaje(*it);
134  }
135 }

265 list<string> Rio::camino(const BinTree<string> &t,
266 const map<string, Ciudad> &info, int &nidc, int &nidv, int idc, int idv)
267 {
268  list<string> ret;
269  // Caso base
270  if (t.empty()) ret = list<string>();
271  else {
272  // Caso general
273  int comprados = nidc;
274  int vendidos = nidv;
275 
276  /* EVALUACION DE PRODUCTOS */
277  auto it = info.find(t.value());
278  if (it->second.existe(idc)) {
279  int tiene1 = it->second.consultar_tiene(idc);
280  int quiere1 = it->second.consultar_quiere(idc);
281  // El barco compra productos
282  if (tiene1 > quiere1 and nidc != 0) {
283  int compra = min(tiene1 - quiere1, nidc);
284  nidc -= compra;
285  }
286  }
287  if (it->second.existe(idv)) {
288  int tiene2 = it->second.consultar_tiene(idv);
289  int quiere2 = it->second.consultar_quiere(idv);
290  // El barco vende productos
291  if (tiene2 < quiere2 and nidv != 0) {
292  int venta = min(quiere2 - tiene2, nidv);
293  nidv -= venta;
294  }
295  }
296 
297  // Caso 1: El barco ya no tiene mas productos (paramos la ejecucion)
298  if (nidc == 0 and nidv == 0) {
299  ret = list<string>(1, t.value());
300  }
301  else {
302  /* LLAMADAS RECURSIVAS */
303  int comp_izq = nidc;
304  int vend_izq = nidv;
305  list<string> izq = camino(t.left(), info, comp_izq, vend_izq, idc,
306  idv);
307  int comp_der = nidc;
308  int vend_der = nidv;
309  list<string> der = camino(t.right(), info, comp_der, vend_der, idc,
310  idv);
311  /* HI: izq es el camino mas provechoso del subarbol izquierdo y
312  der el camino mas provechoso del subarbol derecho
313  */
314 
315  // Caso 2: El nodo es una hoja y no se ha comprado y vendido ningun
316  // producto
317  if (comp_izq == nidc and vend_izq == nidv and comp_der == nidc
318  and vend_der == nidv) {
319  if (comprados == nidc and vendidos == nidv)
320  ret = list<string>();
321  else ret = list<string>(1, t.value());
322  }
323  else {
324  // Caso 3: Ruta izquierda
325  if ((comp_izq + vend_izq) < (comp_der + vend_der)) {
326  izq.push_front(t.value());
327  nidc = comp_izq;
328  nidv = vend_izq;
329  ret = izq;
330  }
331  // Caso 4: Ruta derecha
332  else if (comp_izq + vend_izq > comp_der + vend_der) {
333  der.push_front(t.value());
334  nidc = comp_der;
335  nidv = vend_der;
336  ret = der;
337  }
338  // Caso 5: Depende
339  else {
340  if (izq.size() <= der.size()) {
341  izq.push_front(t.value());
342  nidc = comp_izq;
343  nidv = vend_izq;
344  ret = izq;
345  }
346  else {
347  der.push_front(t.value());
348  nidc = comp_der;
349  nidv = vend_der;
350  ret = der;
351  }
352  }
353  }
354  }
355  }
356  return ret;
357 }
